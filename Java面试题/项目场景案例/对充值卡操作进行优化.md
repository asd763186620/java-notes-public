# 问题
批量新增充值卡、批量导出充值卡、查询充值卡列表接口耗时长

# 原因
串行调用微信生成二维码接口

# 优化方案
批量新增时：并且新增时直接返回生成的充值卡的基本信息，二维码信息异步调用，并且在异步调内部将多个充值由串行调用微信生成二维码接口改为并发执行，并且将成功返回的图片流存入oss，针对失败的充值卡添加重试和熔断机制，单个充值卡重试三次失败则直接将其添加到失败的集合中并且设置状态为失败图片二维码链接为null，成功将状态设置为已生成将oss返回地址赋值给二维码字段中，添加到成功的集合中最后汇总批量修改对应状态和添加二维码信息。如果多个充值卡都失败次数到了5次直接启动熔断机制，暂停微信接口的访问，30秒后可重新访问微信接口，同时通知后端开发或者运维进行排查。

查询充值卡时，直接从数据库中获取对应的二维码oss地址，然后前端进行渲染，同时根据状态的不同展示不同，待生成展示生成中，失败的展示点击重试（后端写单一的请求微信二维码的接口进行重试），成功的直接渲染链接url为图片。

批量导出时，通过懒加载的办法，将失败的没有url链接的充值卡再次进行异步+线程池调用微信接口，获取二维码并且将返回的数据流写入excel，同时此操作也加入重试机制，三次失败则直接返回null，excel不设置url。对于成功已有链接的充值卡通过oss下载对应流写入excel，这一步也需要通过异步编排+线程池操作，同时引入重试机制，对于多次下载失败的url，直接赋值空防止阻塞程序运行

# 核心关键点
completablefuture+线程池异步编排

线程池设置为服务器核心数的2倍（由于微信接口和oss下载都是网络io操作，线程大部分时间用于等待响应，不是占有cpu，因此设置为2n方便cpu处理其他线程）

重试机制3次之后失败

熔断机制多个异步线程调用微信失败对微信接口添加熔断措施，间隔多长时间再恢复

熔断机制次数设计多线程处理使用线程安全的AtomicInteger

同时批量处理的充值卡使用线程安全的Collections.synchronizedList(newArrayList<>());

# 代码实现
```java
package com.lj.doublePointer;

import lombok.extern.slf4j.Slf4j;

import java.io.IOException;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.stream.Collectors;

/**
 * 充值卡服务 - 完整示例
 * 涵盖：异步生成、重试机制、熔断降级、并行导出、超时控制
 */
@Service
@Slf4j
public class RechargeCardService {

    @Autowired
    private ThreadPoolTaskExecutor threadPoolTaskExecutor;

    @Autowired
    private MiniAppConfigService miniAppConfigService;

    @Autowired
    private RechargeCardMapper rechargeCardMapper;

    @Autowired
    private OssService ossService;

    // 熔断状态（生产环境用Resilience4j，这里手写简化版）
    private final AtomicInteger failureCount = new AtomicInteger(0);
    private volatile long circuitOpenTime = 0;
    private static final int FAILURE_THRESHOLD = 5;      // 失败5次触发熔断
    private static final long CIRCUIT_TIMEOUT = 30_000L; // 熔断30秒

    // ==================== 第一阶段：批量新增充值卡 ====================

    /**
     * 批量新增充值卡（异步生成二维码）
     * 立刻返回，后台异步处理二维码生成
     */
    public void batchCreate(List<RechargeCard> cards) {
        // 1. 先保存基本信息，状态标记为"待生成"
        cards.forEach(card -> card.setQrStatus(QrStatusEnum.PENDING.getCode()));
        rechargeCardMapper.batchInsert(cards);

        // 2. 异步生成二维码，不阻塞用户
        String accessToken = miniAppConfigService.getAccessToken();
        CompletableFuture.runAsync(() -> {
            batchGenerateQrCode(cards, accessToken);
        }, threadPoolTaskExecutor);

        // 3. 立刻告诉用户"生成中"
        log.info("充值卡批量新增完成，共{}张，二维码后台生成中", cards.size());
    }

    /**
     * 批量生成二维码（带重试+熔断）
     */
    private void batchGenerateQrCode(List<RechargeCard> cards, String accessToken) {
        List<Long> idsSuccess = Collections.synchronizedList(new ArrayList<>());
        List<Long> idsFail = Collections.synchronizedList(new ArrayList<>());
        List<CompletableFuture<Void>> futures = cards.stream()
                .map(card -> CompletableFuture.runAsync(() -> {
                    Boolean flag = generateSingleQrCode(card, accessToken);
                    if (flag){
                        idsSuccess.add(card.getId());
                    } else {
                        idsFail.add(card.getId());
                    }
                }, threadPoolTaskExecutor))
                .collect(Collectors.toList());

        // 等待全部完成
        CompletableFuture.allOf(futures.toArray(new CompletableFuture[0])).join();
        // 批量处理失败
        rechargeCardMapper.batchUpdateSuccess(idsSuccess);
        rechargeCardMapper.batchUpdateFail(idsFail);

        log.info("二维码批量生成完毕");
    }

    /**
     * 单张二维码生成（带重试+熔断）
     */
    private Boolean generateSingleQrCode(RechargeCard card, String accessToken) {
        // 熔断检查：如果微信接口持续失败，直接跳过不再请求
        if (isCircuitOpen()) {
            log.warn("熔断开启，跳过二维码生成，cardCode={}", card.getCardCode());
            card.setQrStatus(QrStatusEnum.FAILED.getCode());
            // 告警通知（生产环境接钉钉/企微/短信）
            sendAlert("微信接口熔断，请立即排查");
            return false;
        }

        // 重试机制：最多3次
        int maxRetry = 3;
        int attempt = 0;
        byte[] qrCodeData = null;

        while (attempt < maxRetry) {
            try {
                qrCodeData = miniAppConfigService.generateQrCode(
                        "subPackages/changerCenter/changerCenter",
                        "cardCode=" + card.getCardCode(),
                        accessToken
                );
                if (qrCodeData != null && !isErrorResponse(qrCodeData)) {
                    // 成功，重置熔断计数
                    failureCount.set(0);
                    break;
                }
            } catch (Exception e) {
                log.warn("第{}次生成失败，cardCode={}", attempt + 1, card.getCardCode());
            }
            attempt++;
        }

        if (qrCodeData != null && !isErrorResponse(qrCodeData)) {
            // 成功：存OSS，更新数据库
            String ossUrl = ossService.upload(qrCodeData, card.getCardCode() + ".png");
            card.setQrUrl(ossUrl);
            card.setQrStatus(QrStatusEnum.SUCCESS.getCode());
            return true;
        } else {
            // 3次全部失败：标记失败，触发熔断计数
            card.setQrStatus(QrStatusEnum.FAILED.getCode());
            int failures = failureCount.incrementAndGet();
            if (failures >= FAILURE_THRESHOLD) {
                // 触发熔断
                circuitOpenTime = System.currentTimeMillis();
                log.error("微信接口连续失败{}次，熔断开启", failures);
                sendAlert("微信接口熔断触发，失败次数：" + failures);
            }
            return false;
        }
    }

    // ==================== 第二阶段：列表查询 ====================

    /**
     * 查询充值卡列表（直接读OSS地址，不实时请求微信）
     */
    public List<RechargeCardVO> listCards(RechargeCardQuery query) {
        List<RechargeCard> cards = rechargeCardMapper.selectByQuery(query);
        return cards.stream().map(card -> {
            RechargeCardVO vo = new RechargeCardVO();
            BeanUtils.copyProperties(card, vo);
            // 直接返回OSS地址，前端用链接渲染图片
            vo.setQrUrl(card.getQrUrl());
            vo.setQrStatus(card.getQrStatus()); // 前端根据状态显示"生成失败"或正常图片
            return vo;
        }).collect(Collectors.toList());
    }

    // ==================== 第三阶段：批量导出Excel ====================

    /**
     * 导出充值卡Excel（并行下载OSS图片+超时控制+失败补偿）
     */
    public void exportExcel(RechargeCardQuery query, HttpServletResponse response) {
        List<RechargeCard> cards = rechargeCardMapper.selectByQuery(query);
        if (CollectionUtil.isEmpty(cards)) {
            throw new BusinessException("无数据可导出");
        }

        String accessToken = miniAppConfigService.getAccessToken();
        AtomicInteger index = new AtomicInteger(1);

        // 并行处理每一条（下载OSS图片+失败补偿）
        List<CompletableFuture<RechargeCardExcelVO>> futures = cards.stream()
                .map(card -> CompletableFuture
                        .supplyAsync(() -> buildExcelVO(card, index, accessToken), threadPoolTaskExecutor)
                        .orTimeout(5, TimeUnit.SECONDS)          // 超时5秒不等了
                        .exceptionally(e -> {
                            // 超时或异常：返回无二维码的行，不阻塞整体导出
                            log.warn("导出时获取二维码超时，cardCode={}", card.getCardCode());
                            return buildExcelVOWithoutQr(card, index);
                        })
                )
                .collect(Collectors.toList());

        // 等待全部完成，聚合结果
        List<RechargeCardExcelVO> exportList = futures.stream()
                .map(CompletableFuture::join)
                .collect(Collectors.toList());

        // 写入Excel返回
        writeExcel(response, exportList);
    }

    /**
     * 构建单行Excel数据（含失败补偿：OSS为空则实时请求微信）
     */
    private RechargeCardExcelVO buildExcelVO(RechargeCard card, AtomicInteger index, String accessToken) {
        RechargeCardExcelVO vo = new RechargeCardExcelVO();
        vo.setIndex(index.getAndIncrement());
        vo.setCardName(card.getCardName());
        vo.setCardCode(card.getCardCode());
        vo.setCardPassword(card.getCardPassword());

        byte[] imageData = null;

        if (StringUtils.isNotBlank(card.getQrUrl())) {
            // 正常情况：从OSS下载
            imageData = ossService.download(card.getQrUrl());
        } else {
            // 失败补偿：OSS地址为空，实时请求微信接口补生成
            log.info("导出时发现二维码缺失，实时补偿，cardCode={}", card.getCardCode());
            imageData = miniAppConfigService.generateQrCode(
                    "subPackages/changerCenter/changerCenter",
                    "cardCode=" + card.getCardCode(),
                    accessToken
            );
            if (imageData != null && !isErrorResponse(imageData)) {
                // 补偿成功，顺手存OSS更新数据库
                String ossUrl = ossService.upload(imageData, card.getCardCode() + ".png");
                card.setQrUrl(ossUrl);
                card.setQrStatus(QrStatusEnum.SUCCESS.getCode());
                rechargeCardMapper.updateById(card);
            }
        }

        if (imageData != null && !isErrorResponse(imageData)) {
            vo.setQrCodeImage(imageData);
        }
        return vo;
    }

    /**
     * 构建无二维码的行（超时兜底）
     */
    private RechargeCardExcelVO buildExcelVOWithoutQr(RechargeCard card, AtomicInteger index) {
        RechargeCardExcelVO vo = new RechargeCardExcelVO();
        vo.setIndex(index.getAndIncrement());
        vo.setCardName(card.getCardName());
        vo.setCardCode(card.getCardCode());
        vo.setCardPassword(card.getCardPassword());
        vo.setQrCodeImage(null); // 二维码为空，Excel该行无图片
        return vo;
    }

    // ==================== 工具方法 ====================

    /**
     * 熔断检查：是否处于熔断状态
     */
    private boolean isCircuitOpen() {
        if (failureCount.get() < FAILURE_THRESHOLD) return false;
        // 熔断超时后自动恢复（半开状态）
        if (System.currentTimeMillis() - circuitOpenTime > CIRCUIT_TIMEOUT) {
            failureCount.set(0);
            log.info("熔断恢复，重新尝试请求微信接口");
            return false;
        }
        return true;
    }

    private boolean isErrorResponse(byte[] data) {
        // 微信接口错误时返回JSON，正常时返回图片二进制
        String str = new String(data, 0, Math.min(data.length, 20));
        return str.contains("{");
    }

    private void sendAlert(String message) {
        // 生产环境接入钉钉/企微机器人/短信告警
        log.error("[告警] {}", message);
    }

    private void writeExcel(HttpServletResponse response, List<RechargeCardExcelVO> data) {
        try {
            response.setContentType("application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");
            response.setHeader("Content-disposition", "attachment;filename=RechargeCard.xlsx");
            EasyExcel.write(response.getOutputStream(), RechargeCardExcelVO.class)
                    .sheet("充值卡")
                    .doWrite(data);
        } catch (IOException e) {
            throw new BusinessException("导出失败");
        }
    }
}
```

# 面试一句话总结
通过completablefuture+线程池异步编排+oss存储方式优化充值卡接口二维码生成问题，将充值卡新增和导出接口耗时从分钟优化到秒，通过预生成的方式将微信的依赖降到最小，同时添加重试和熔断机制以及告警，保护系统运行方便处理程序错误，给用户好的使用体验。

充值卡导出从分钟到秒，CompletableFuture+线程池并行化，预生成存OSS消除微信依赖，重试+熔断+告警保障稳定性
